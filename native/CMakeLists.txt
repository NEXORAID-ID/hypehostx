cmake_minimum_required(VERSION 3.24)
project(HYPEHENNative VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(hypecore src/core.cpp src/catalog.cpp)
target_include_directories(hypecore PUBLIC include)
target_include_directories(hypecore SYSTEM PRIVATE third_party)
option(HYPE_SANITIZE "Address sanitizer for portable development only" OFF)
if(HYPE_SANITIZE)
 if(MSVC)
  target_compile_options(hypecore PUBLIC /fsanitize=address)
  target_link_options(hypecore PUBLIC /INCREMENTAL:NO)
 else()
  target_compile_options(hypecore PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(hypecore PUBLIC -fsanitize=address,undefined)
 endif()
endif()
if(MSVC)
 target_compile_options(hypecore PRIVATE /W4 /WX /EHsc)
else()
 target_compile_options(hypecore PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(hypehen src/main.cpp)
target_link_libraries(hypehen PRIVATE hypecore)
add_executable(native_tests tests/core_test.cpp)
target_link_libraries(native_tests PRIVATE hypecore)
add_executable(hypekit_sample ../packages/hypekit/sample.cpp)
target_link_libraries(hypekit_sample PRIVATE hypecore)
enable_testing()
add_test(NAME portable_domain COMMAND native_tests)
add_test(NAME extension_sample COMMAND hypekit_sample)
